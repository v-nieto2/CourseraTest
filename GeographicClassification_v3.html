from shapely.geometry import box, mapping

with rasterio.open(land_tif) as src:
    RASTER_CRS = src.crs

    print("Raster CRS:", RASTER_CRS)
    print("Raster resolution:", src.res)
    print("Raster bounds:", src.bounds)

    # Project quarter sections into raster CRs
    qtr_section_gdf = gpd.read_file(qtr_section)
    aoi_raster_crs = qtr_section_gdf.to_crs(RASTER_CRS)


    # Use the rectangular operating-area extent instead of unioning polygons
    minx, miny, maxx, maxy = (
        aoi_raster_crs.total_bounds
    )

    aoi_geometry = box(
        minx,
        miny,
        maxx,
        maxy
    ).buffer(50000)   # 10 km

        src,
        [mapping(aoi_geometry)],
        crop=True,

    )

# This treats SMOOTHING_RADIUS_M approximately as a 2-sigma neighborhood.

    SMOOTHING_RADIUS_M / pixel_width_m

def save_classified_raster(
    array,
    output_path,
    metadata
):
    output_meta = metadata.copy()
    output_meta.update(
        dtype="uint8",
        nodata=0,
        compress="lzw"
    )

    with rasterio.open(
        output_path,
        "w",
        **output_meta
    ) as dst:
        dst.write(
            array.astype(np.uint8),
            1
        )


def save_float_raster(
    array,
    output_path,
    metadata
):
    output_meta = metadata.copy()
    output_meta.update(
        dtype="float32",
        nodata=-9999,
        compress="lzw"
    )

    output = np.where(
        np.isfinite(array),
        array,
        -9999
    ).astype(np.float32)

    with rasterio.open(
        output_path,
        "w",
        **output_meta
    ) as dst:
        dst.write(output, 1)

# =============================================================================
# ASSIGN THREE RASTER-BASED AREA CLASSIFICATIONS TO BAY SPANS
#
# Method:
#   1. Correct bay geometries to EPSG:4326 if currently mislabeled.
#   2. Reproject each bay to the CRS of each classified raster.
#   3. Sample evenly spaced points along each span.
#   4. Assign the tier receiving the majority of valid raster samples.
#   5. Break ties using the midpoint sample, then Urban > Suburban > Rural.
#
# Expected raster coding:
#   0 = NoData
#   1 = Rural
#   2 = Suburban
#   3 = Urban
#
# Output:
#   area_lc              Majority tier from land-cover-only raster
#   area_lc_pct          % of valid samples supporting area_lc
#   area_lc_n_valid      Number of valid raster samples
#   area_lc_source       Assignment method
#
#   area_lcxfm
#   area_lcxfm_pct
#   area_lcxfm_n_valid
#   area_lcxfm_source
#
#   area_xfmr
#   area_xfmr_pct
#   area_xfmr_n_valid
#   area_xfmr_source
#
# bay remains both a column and the index.
# =============================================================================

import os
import numpy as np
import pandas as pd
import geopandas as gpd
import rasterio

from rasterio.transform import rowcol
from shapely.geometry import Point


# =============================================================================
# CONFIGURATION
# =============================================================================

OUTPUT_DIR = "./deployment/area_type_polygons"

LAND_COVER_ONLY_RASTER = os.path.join(
    OUTPUT_DIR,
    "land_cover_only.tif"
)

XFMR_SUPPORTED_RASTER = os.path.join(
    OUTPUT_DIR,
    "xfmr_supported_land_cover.tif"
)

XFMR_ONLY_RASTER = os.path.join(
    OUTPUT_DIR,
    "xfmr_only.tif"
)

# Eleven interior points provide a reasonable balance between speed and
# representation of the span. Increase to 21 if spans frequently cross
# several land-cover boundaries.
N_SAMPLE_POINTS = 11

# Avoid sampling exactly at line endpoints because endpoints may fall
# just outside the classified raster or on a NoData boundary.
SAMPLE_FRACTIONS = np.linspace(
    0.05,
    0.95,
    N_SAMPLE_POINTS
)

# Raster class definitions
TIER_LOOKUP = {
    1: "Rural",
    2: "Suburban",
    3: "Urban"
}

# Used only after a majority-vote tie that cannot be resolved by midpoint.
TIER_PRIORITY = {
    1: 1,  # Rural
    2: 2,  # Suburban
    3: 3   # Urban
}

VALID_RASTER_CODES = set(TIER_LOOKUP.keys())


# =============================================================================
# HELPER FUNCTIONS
# =============================================================================

def interpolate_point_safe(geometry, fraction):
    """
    Return a point at a normalized fraction along a LineString or
    MultiLineString.

    Returns None for missing, empty, zero-length, or invalid geometries.
    """

    if geometry is None or geometry.is_empty:
        return None

    if geometry.geom_type not in {
        "LineString",
        "MultiLineString"
    }:
        return None

    if geometry.length <= 0:
        return None

    try:
        return geometry.interpolate(
            fraction,
            normalized=True
        )
    except Exception:
        return None


def determine_majority_tier(
    sampled_codes,
    midpoint_position
):
    """
    Determine the majority raster class for one span.

    Tie-breaking:
      1. Use the class sampled at the midpoint.
      2. If midpoint is NoData or not one of the tied classes,
         use Urban > Suburban > Rural.

    Parameters
    ----------
    sampled_codes : list
        Raster values sampled along one span.

    midpoint_position : int
        Sample-list position closest to 0.50.

    Returns
    -------
    assigned_code : int or None
    majority_pct : float
    valid_count : int
    assignment_source : str
    """

    valid_codes = [
        int(code)
        for code in sampled_codes
        if pd.notna(code)
        and int(code) in VALID_RASTER_CODES
    ]

    valid_count = len(valid_codes)

    if valid_count == 0:
        return None, 0.0, 0, "no_valid_raster_samples"

    counts = pd.Series(
        valid_codes
    ).value_counts()

    maximum_votes = counts.max()

    winning_codes = sorted(
        counts[
            counts == maximum_votes
        ].index.tolist()
    )

    # Clear majority or plurality winner
    if len(winning_codes) == 1:
        assigned_code = int(winning_codes[0])
        assignment_source = "sample_majority"

    else:
        # Try the midpoint sample first
        midpoint_code = sampled_codes[
            midpoint_position
        ]

        if (
            pd.notna(midpoint_code)
            and int(midpoint_code) in winning_codes
        ):
            assigned_code = int(midpoint_code)
            assignment_source = "sample_tie_midpoint"

        else:
            # Final deterministic tie-breaker
            assigned_code = max(
                winning_codes,
                key=lambda code: TIER_PRIORITY[int(code)]
            )

            assigned_code = int(assigned_code)
            assignment_source = "sample_tie_priority"

    majority_pct = (
        100.0
        * counts.loc[assigned_code]
        / valid_count
    )

    return (
        assigned_code,
        round(majority_pct, 2),
        valid_count,
        assignment_source
    )


def assign_raster_tier_to_spans(
    spans_gdf,
    raster_path,
    output_column,
    bay_column="bay",
    sample_fractions=SAMPLE_FRACTIONS,
    verbose=True
):
    """
    Sample a classified raster along each span and assign the majority tier.

    The percentage output is the percentage of valid sampled points supporting
    the selected tier. It is an approximation of span coverage, not an exact
    line-intersection percentage.
    """

    # -------------------------------------------------------------------------
    # Validate inputs
    # -------------------------------------------------------------------------

    if not os.path.exists(raster_path):
        raise FileNotFoundError(
            f"Raster was not found: {raster_path}"
        )

    if not isinstance(
        spans_gdf,
        gpd.GeoDataFrame
    ):
        raise TypeError(
            "spans_gdf must be a GeoDataFrame."
        )

    if bay_column not in spans_gdf.columns:
        raise KeyError(
            f"'{bay_column}' was not found in spans_gdf. "
            f"Available columns: {spans_gdf.columns.tolist()}"
        )

    if spans_gdf.crs is None:
        raise ValueError(
            "spans_gdf has no assigned CRS."
        )

    spans = spans_gdf.copy()

    # Standardize active geometry name
    if spans.geometry.name != "geometry":
        spans = spans.rename_geometry(
            "geometry"
        )

    # Retain only usable bay IDs
    spans = spans[
        spans[bay_column].notna()
    ].copy()

    spans[bay_column] = (
        spans[bay_column]
        .astype(str)
        .str.strip()
    )

    spans = spans[
        spans[bay_column] != ""
    ].copy()

    duplicate_bays = spans.loc[
        spans[bay_column].duplicated(
            keep=False
        ),
        bay_column
    ].unique()

    if len(duplicate_bays) > 0:
        raise ValueError(
            f"The '{bay_column}' field must uniquely identify each span. "
            f"Found {len(duplicate_bays):,} duplicated values. "
            f"Examples: {duplicate_bays[:10].tolist()}"
        )

    # Set bay as index while preserving the column
    spans = spans.set_index(
        bay_column,
        drop=False
    )

    # Prepare output fields for every bay, including invalid geometries
    percentage_column = (
        f"{output_column}_pct"
    )

    valid_sample_column = (
        f"{output_column}_n_valid"
    )

    source_column = (
        f"{output_column}_source"
    )

    spans[output_column] = "Unknown"
    spans[percentage_column] = 0.0
    spans[valid_sample_column] = 0
    spans[source_column] = (
        "invalid_or_unsupported_geometry"
    )

    valid_geometry_mask = (
        spans.geometry.notna()
        & ~spans.geometry.is_empty
        & spans.geometry.geom_type.isin(
            ["LineString", "MultiLineString"]
        )
        & (spans.geometry.length > 0)
    )

    valid_spans = spans.loc[
        valid_geometry_mask
    ].copy()

    if valid_spans.empty:
        if verbose:
            print(
                f"{output_column}: no valid span geometries found."
            )

        return spans

    # -------------------------------------------------------------------------
    # Read raster and project spans into raster CRS
    # -------------------------------------------------------------------------

    with rasterio.open(
        raster_path
    ) as src:

        if src.crs is None:
            raise ValueError(
                f"Raster has no CRS: {raster_path}"
            )

        raster_crs = src.crs
        raster_nodata = src.nodata
        raster_bounds = src.bounds

        # Transform actual coordinates into raster CRS
        valid_spans_raster = (
            valid_spans.to_crs(raster_crs)
        )

        span_bounds = (
            valid_spans_raster.total_bounds
        )

        span_minx, span_miny, span_maxx, span_maxy = (
            span_bounds
        )

        bounds_overlap = not (
            span_maxx < raster_bounds.left
            or span_minx > raster_bounds.right
            or span_maxy < raster_bounds.bottom
            or span_miny > raster_bounds.top
        )

        if verbose:
            print(
                f"\nProcessing {output_column}"
            )
            print(
                "Raster:",
                raster_path
            )
            print(
                "Raster CRS:",
                raster_crs
            )
            print(
                "Raster resolution:",
                src.res
            )
            print(
                "Raster NoData:",
                raster_nodata
            )
            print(
                "Raster bounds:",
                raster_bounds
            )
            print(
                "Projected span bounds:",
                span_bounds
            )
            print(
                "Bounds overlap:",
                bounds_overlap
            )
            print(
                "Valid span count:",
                len(valid_spans_raster)
            )
            print(
                "Samples per span:",
                len(sample_fractions)
            )

        if not bounds_overlap:
            raise ValueError(
                f"The spans do not overlap {raster_path} after "
                "reprojection. Check the source CRS assigned to bay_data."
            )

        # Read once into memory. These classification rasters are clipped
        # to the operating area and contain only one uint8 band.
        raster_array = src.read(1)

        raster_height = src.height
        raster_width = src.width
        raster_transform = src.transform

        # Sample closest point to exactly 50% for tie-breaking
        midpoint_position = int(
            np.argmin(
                np.abs(
                    np.asarray(sample_fractions)
                    - 0.50
                )
            )
        )

        # ---------------------------------------------------------------------
        # Build all sample coordinates
        # ---------------------------------------------------------------------

        sample_records = []

        for bay_value, geometry in (
            valid_spans_raster.geometry.items()
        ):

            for sample_number, fraction in enumerate(
                sample_fractions
            ):
                sample_point = interpolate_point_safe(
                    geometry,
                    float(fraction)
                )

                if sample_point is None:
                    sample_records.append({
                        bay_column: bay_value,
                        "sample_number": sample_number,
                        "fraction": float(fraction),
                        "x": np.nan,
                        "y": np.nan
                    })

                else:
                    sample_records.append({
                        bay_column: bay_value,
                        "sample_number": sample_number,
                        "fraction": float(fraction),
                        "x": sample_point.x,
                        "y": sample_point.y
                    })

        sample_df = pd.DataFrame(
            sample_records
        )

        # ---------------------------------------------------------------------
        # Convert coordinates to raster row and column indices
        # ---------------------------------------------------------------------

        sample_df["raster_code"] = np.nan

        coordinate_mask = (
            sample_df["x"].notna()
            & sample_df["y"].notna()
        )

        coordinate_rows = sample_df.loc[
            coordinate_mask
        ].copy()

        if not coordinate_rows.empty:
            rows, cols = rowcol(
                raster_transform,
                coordinate_rows["x"].to_numpy(),
                coordinate_rows["y"].to_numpy()
            )

            coordinate_rows["row"] = np.asarray(
                rows,
                dtype=np.int64
            )

            coordinate_rows["col"] = np.asarray(
                cols,
                dtype=np.int64
            )

            # Explicit bounds test prevents accidental negative indexing
            inside_raster = (
                coordinate_rows["row"].between(
                    0,
                    raster_height - 1
                )
                & coordinate_rows["col"].between(
                    0,
                    raster_width - 1
                )
            )

            inside_rows = coordinate_rows.loc[
                inside_raster
            ].copy()

            if not inside_rows.empty:
                sampled_values = raster_array[
                    inside_rows["row"].to_numpy(
                        dtype=np.int64
                    ),
                    inside_rows["col"].to_numpy(
                        dtype=np.int64
                    )
                ]

                sampled_values = sampled_values.astype(
                    np.float64
                )

                # Convert raster NoData to NaN
                if raster_nodata is not None:
                    sampled_values[
                        sampled_values == raster_nodata
                    ] = np.nan

                # Classification code 0 is NoData
                sampled_values[
                    sampled_values == 0
                ] = np.nan

                # Reject unexpected raster values
                expected_code_mask = np.isin(
                    sampled_values,
                    list(VALID_RASTER_CODES)
                )

                sampled_values[
                    ~expected_code_mask
                    & np.isfinite(sampled_values)
                ] = np.nan

                sample_df.loc[
                    inside_rows.index,
                    "raster_code"
                ] = sampled_values

        # ---------------------------------------------------------------------
        # Majority vote by bay
        # ---------------------------------------------------------------------

        assignment_records = []

        for bay_value, bay_samples in sample_df.groupby(
            bay_column,
            sort=False
        ):
            bay_samples = bay_samples.sort_values(
                "sample_number"
            )

            sampled_codes = (
                bay_samples["raster_code"]
                .tolist()
            )

            (
                assigned_code,
                majority_pct,
                valid_count,
                assignment_source
            ) = determine_majority_tier(
                sampled_codes=sampled_codes,
                midpoint_position=midpoint_position
            )

            assigned_tier = (
                TIER_LOOKUP.get(
                    assigned_code,
                    "Unknown"
                )
            )

            assignment_records.append({
                bay_column: bay_value,
                output_column: assigned_tier,
                percentage_column: majority_pct,
                valid_sample_column: valid_count,
                source_column: assignment_source
            })

        assignment_df = (
            pd.DataFrame(
                assignment_records
            )
            .set_index(bay_column)
        )

    # -------------------------------------------------------------------------
    # Assign results back to the complete span layer
    # -------------------------------------------------------------------------

    matching_bays = spans.index.intersection(
        assignment_df.index
    )

    for column in [
        output_column,
        percentage_column,
        valid_sample_column,
        source_column
    ]:
        spans.loc[
            matching_bays,
            column
        ] = assignment_df.loc[
            matching_bays,
            column
        ]

    spans[output_column] = (
        spans[output_column]
        .fillna("Unknown")
    )

    spans[percentage_column] = (
        pd.to_numeric(
            spans[percentage_column],
            errors="coerce"
        )
        .fillna(0)
        .clip(lower=0, upper=100)
        .round(2)
    )

    spans[valid_sample_column] = (
        pd.to_numeric(
            spans[valid_sample_column],
            errors="coerce"
        )
        .fillna(0)
        .astype(int)
    )

    spans[source_column] = (
        spans[source_column]
        .fillna("no_valid_raster_samples")
    )

    spans.index.name = bay_column
    spans[bay_column] = spans.index.astype(str)

    if verbose:
        print("\nAssigned tier counts:")
        print(
            spans[output_column]
            .value_counts(dropna=False)
        )

        print("\nAssignment source:")
        print(
            spans[source_column]
            .value_counts(dropna=False)
        )

        print("\nValid raster samples:")
        print(
            spans[valid_sample_column]
            .describe()
        )

        print("\nMajority support percentage:")
        print(
            spans[percentage_column]
            .describe()
        )

    return spans
